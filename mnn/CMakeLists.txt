cmake_minimum_required(VERSION 3.18)
project(mnn LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(MNN_CUDA "Compila o backend CUDA" OFF)

add_library(mnn STATIC src/backend.cpp src/cpu_backend.cpp)
target_include_directories(mnn PUBLIC include)
if(MNN_CUDA)
  enable_language(CUDA)
  target_sources(mnn PRIVATE src/cuda_backend.cu)
  target_compile_definitions(mnn PUBLIC MNN_WITH_CUDA)
  set_target_properties(mnn PROPERTIES CUDA_ARCHITECTURES native)
endif()

add_executable(test_all tests/test_all.cpp)
target_link_libraries(test_all mnn)
add_executable(spiral examples/spiral.cpp)
target_include_directories(spiral PRIVATE examples)
target_link_libraries(spiral mnn)
enable_testing()
add_test(NAME cpu COMMAND test_all cpu)
